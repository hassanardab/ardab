=== ./scripts/reset-project.js ===
#!/usr/bin/env node

/**
 * This script is used to reset the project to a blank state.
 * It deletes or moves the /src and /scripts directories to /example based on user input and creates a new /src/app directory with an index.tsx and _layout.tsx file.
 * You can remove the `reset-project` script from package.json and safely delete this file after running it.
 */

const fs = require("fs");
const path = require("path");
const readline = require("readline");

const root = process.cwd();
const oldDirs = ["src", "scripts"];
const exampleDir = "example";
const newAppDir = "src/app";
const exampleDirPath = path.join(root, exampleDir);

const indexContent = `import { Text, View, StyleSheet } from "react-native";

export default function Index() {
  return (
    <View style={styles.container}>
      <Text>Edit src/app/index.tsx to edit this screen.</Text>
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    alignItems: "center",
    justifyContent: "center",
  },
});
`;

const layoutContent = `import { Stack } from "expo-router";

export default function RootLayout() {
  return <Stack />;
}
`;

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
});

const moveDirectories = async (userInput) => {
  try {
    if (userInput === "y") {
      // Create the app-example directory
      await fs.promises.mkdir(exampleDirPath, { recursive: true });
      console.log(`📁 /${exampleDir} directory created.`);
    }

    // Move old directories to new app-example directory or delete them
    for (const dir of oldDirs) {
      const oldDirPath = path.join(root, dir);
      if (fs.existsSync(oldDirPath)) {
        if (userInput === "y") {
          const newDirPath = path.join(root, exampleDir, dir);
          await fs.promises.rename(oldDirPath, newDirPath);
          console.log(`➡️ /${dir} moved to /${exampleDir}/${dir}.`);
        } else {
          await fs.promises.rm(oldDirPath, { recursive: true, force: true });
          console.log(`❌ /${dir} deleted.`);
        }
      } else {
        console.log(`➡️ /${dir} does not exist, skipping.`);
      }
    }

    // Create new /src/app directory
    const newAppDirPath = path.join(root, newAppDir);
    await fs.promises.mkdir(newAppDirPath, { recursive: true });
    console.log("\n📁 New /src/app directory created.");

    // Create index.tsx
    const indexPath = path.join(newAppDirPath, "index.tsx");
    await fs.promises.writeFile(indexPath, indexContent);
    console.log("📄 src/app/index.tsx created.");

    // Create _layout.tsx
    const layoutPath = path.join(newAppDirPath, "_layout.tsx");
    await fs.promises.writeFile(layoutPath, layoutContent);
    console.log("📄 src/app/_layout.tsx created.");

    console.log("\n✅ Project reset complete. Next steps:");
    console.log(
      `1. Run \`npx expo start\` to start a development server.\n2. Edit src/app/index.tsx to edit the main screen.\n3. Put all your application code in /src, only screens and layout files should be in /src/app.${
        userInput === "y"
          ? `\n4. Delete the /${exampleDir} directory when you're done referencing it.`
          : ""
      }`
    );
  } catch (error) {
    console.error(`❌ Error during script execution: ${error.message}`);
  }
};

rl.question(
  "Do you want to move existing files to /example instead of deleting them? (Y/n): ",
  (answer) => {
    const userInput = answer.trim().toLowerCase() || "y";
    if (userInput === "y" || userInput === "n") {
      moveDirectories(userInput).finally(() => rl.close());
    } else {
      console.log("❌ Invalid input. Please enter 'Y' or 'N'.");
      rl.close();
    }
  }
);

// src/app/_layout.tsx
import { MaterialIcons } from "@expo/vector-icons";
import { Tabs } from "expo-router";
import { StatusBar } from "expo-status-bar";
import {
  SafeAreaProvider,
  useSafeAreaInsets,
} from "react-native-safe-area-context";

// NOTE: SafeAreaProvider MUST be above the Tabs so that
// useSafeAreaInsets() works inside every screen.
export default function TabLayout() {
  return (
    <SafeAreaProvider>
      <StatusBar style="dark" />
      <TabsShell />
    </SafeAreaProvider>
  );
}

function TabsShell() {
  const insets = useSafeAreaInsets();

  // Base height the tab bar needs for its icons + labels.
  // (React Navigation's default is ~49 on iOS, ~56 on Android.)
  // We ADD the bottom safe-area inset so the bar's background
  // extends behind the home indicator on modern phones, and add
  // matching bottom padding so the icons stay above it.
  const BASE_TAB_BAR_HEIGHT = 56;
  const TAB_BAR_HEIGHT = BASE_TAB_BAR_HEIGHT + insets.bottom;

  return (
    <Tabs
      screenOptions={{
        headerStyle: { backgroundColor: "#ffffff" },
        headerTintColor: "#000000",
        tabBarActiveTintColor: "#000000",
        tabBarInactiveTintColor: "#a7a9ac",
        tabBarHideOnKeyboard: true,
        tabBarStyle: {
          backgroundColor: "#ffffff",
          borderTopColor: "#a7a9ac",
          borderTopWidth: 0.5,
          // This is what removes the gray strip:
          height: TAB_BAR_HEIGHT,
          paddingBottom: insets.bottom > 0 ? insets.bottom : 6,
          paddingTop: 4,
          // Make sure the background paints even in the inset zone:
          position: "absolute",
          bottom: 0,
          left: 0,
          right: 0,
        },
      }}
    >
      <Tabs.Screen
        name="index"
        options={{
          title: "الرئيسية",
          headerShown: false,
          tabBarIcon: ({ color }) => (
            <MaterialIcons name="dashboard" size={24} color={color} />
          ),
        }}
      />
      <Tabs.Screen
        name="invoices"
        options={{
          title: "الفواتير",
          headerShown: false,
          tabBarIcon: ({ color }) => (
            <MaterialIcons name="receipt" size={24} color={color} />
          ),
        }}
      />
      <Tabs.Screen
        name="messages"
        options={{
          title: "الأسماء",
          headerShown: false,
          tabBarIcon: ({ color }) => (
            <MaterialIcons name="people" size={24} color={color} />
          ),
        }}
      />
      <Tabs.Screen name="project/[id]" options={{ href: null }} />
      <Tabs.Screen
        name="project/[id]/report"
        options={{ href: null, headerShown: false }}
      />
      <Tabs.Screen name="invoice/[id]" options={{ href: null }} />
      <Tabs.Screen name="project/[id]/settings" options={{ href: null }} />
    </Tabs>
  );
}

  const BOTTOM_RESERVED = TAB_BAR_HEIGHT + insets.bottom;

  // ---------- Date range filter state ----------
  const [dateFilter, setDateFilter] = useState<DateFilterType>("ALL");
  const [customStart, setCustomStart] = useState<Date | null>(null);
  const [customEnd, setCustomEnd] = useState<Date | null>(null);
  const [showStartPicker, setShowStartPicker] = useState(false);
  const [showEndPicker, setShowEndPicker] = useState(false);

  // ---------- Handlers ----------
  const handleCreateProject = () => {
    if (!newProjectName) return;
    addProject({
      id: Date.now().toString(),
      name: newProjectName,
      initialCash: 0,
      initialBank: 0,
      createdAt: new Date().toISOString(),
    });
    setNewProjectName("");
  };

  const handleEditProject = (project: Project) => {
    setEditingProject(project);
    setEditName(project.name);
  };

  const saveEdit = () => {
    if (editingProject && editName.trim()) {
      updateProject(editingProject.id, { name: editName.trim() });
      setEditingProject(null);
      setEditName("");
    }
  };

  // ---------- Delete project (with confirmation) ----------
  const handleDeleteProject = () => {
    if (!editingProject) return;
    const projectToDelete = editingProject;
    const txCount = transactions.filter(
      (t) => t.projectId === projectToDelete.id,
    ).length;

    Alert.alert(
      "حذف المشروع",
      `هل أنت متأكد من حذف "${projectToDelete.name}"؟${
        txCount > 0 ? `\nسيتم حذف ${txCount} معاملة مرتبطة به.` : ""
      }`,
      [
        { text: "إلغاء", style: "cancel" },
        {
          text: "حذف",
          style: "destructive",
          onPress: () => {
            removeProject(projectToDelete.id);
            setEditingProject(null);
            setEditName("");
          },
        },
      ],
      { cancelable: true },
    );
  };

  const openTransactionModal = (projectId: string) => {
    setSelectedProjectId(projectId);
    setIsModalVisible(true);
  };

  // ---------- Date range helpers ----------
  const getRange = (): { start: number; end: number } => {
    const now = new Date();
    switch (dateFilter) {
      case "THIS_MONTH":
        return {
          start: new Date(now.getFullYear(), now.getMonth(), 1).getTime(),
          end: new Date(
            now.getFullYear(),
            now.getMonth() + 1,
            0,
            23,
            59,
            59,
            999,
          ).getTime(),
        };
      case "LAST_MONTH":
        return {
          start: new Date(now.getFullYear(), now.getMonth() - 1, 1).getTime(),
          end: new Date(
            now.getFullYear(),
            now.getMonth(),
            0,
            23,
            59,
            59,
            999,
          ).getTime(),
        };
      case "THIS_YEAR":
        return {
          start: new Date(now.getFullYear(), 0, 1).getTime(),
          end: new Date(now.getFullYear(), 11, 31, 23, 59, 59, 999).getTime(),
        };
      case "CUSTOM": {
        const start = customStart ? customStart.getTime() : 0;
        const end = customEnd
          ? new Date(
              customEnd.getFullYear(),
              customEnd.getMonth(),
              customEnd.getDate(),
              23,
              59,
              59,
              999,
            ).getTime()
          : Infinity;
        return { start, end };
      }
      case "ALL":
      default:
        return { start: 0, end: Infinity };
    }
  };

  // Compute all project balances in one pass, keyed by projectId.
  // Direction comes from each project's own transactionTypes, so custom
  // and archived types are handled consistently with the store.
  const projectBalances = useMemo(() => {
    const { start, end } = getRange();
    const result: Record<
      string,
      { cash: number; bank: number; loanBalance: number }
    > = {};

    projects.forEach((project) => {
      let cash = project.initialCash;
      let bank = project.initialBank;
      let loanBalance = 0;

      transactions
        .filter((t) => t.projectId === project.id)
        .forEach((t) => {

          if (ts < start || ts > end) return;

          const def = resolveType(project, t.type);
          // Unknown or informational-only type: no balance effect.
          if (!def || !def.affectsBalance) return;

          if (def.id === "TRANSFER") {
            if (t.method === "BANK") {
              bank += t.amount;
              cash -= t.amount;
            } else {
              cash += t.amount;
              bank -= t.amount;
            }
          } else {
            const isInflow = def.direction === "INFLOW";
            const delta = isInflow ? t.amount : -t.amount;
            if (t.method === "CASH") cash += delta;
            else bank += delta;
          }

          if (def.countsAsLoanReceived) loanBalance += t.amount;
          if (def.countsAsLoanGiven) loanBalance -= t.amount;
        });

      result[project.id] = { cash, bank, loanBalance };
    });

    return result;
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, [projects, transactions, dateFilter, customStart, customEnd]);

    if (dateFilter === "THIS_YEAR") return "هذا العام";

  };

    if (Platform.OS === "android") setShowEndPicker(false);
    if (selectedDate) setCustomEnd(selectedDate);
  };

  // Format today's date in Arabic
  const formattedDate = new Intl.DateTimeFormat("ar-EG", {
    weekday: "long",
    day: "numeric",
    month: "long",
  }).format(new Date());

        style={styles.keyboardContainer}
        behavior={Platform.OS === "ios" ? "padding" : "padding"}
        keyboardVerticalOffset={Platform.OS === "ios" ? 0 : 24}
      >
        <ScrollView
          style={styles.container}
          contentContainerStyle={[
            styles.contentContainer,
            { paddingBottom: 24 + BOTTOM_RESERVED },

          showsVerticalScrollIndicator={false}
          keyboardShouldPersistTaps="handled"
          keyboardDismissMode={
            Platform.OS === "ios" ? "interactive" : "on-drag"
          }
          automaticallyAdjustKeyboardInsets
        >
          {/* Top Header Bar with Date */}

            />
          </View>

          <View style={styles.filterCard}>
            <View style={styles.filterHeaderRow}>
              <Text style={styles.filterPeriodLabel}>{periodLabel}</Text>
              <Text style={styles.filterTitle}>الفترة الزمنية</Text>
            </View>

            <ScrollView
              horizontal
              showsHorizontalScrollIndicator={false}
              contentContainerStyle={styles.filterChipsScroll}
            >
              {(
                [
                  { id: "ALL", label: "الكل" },
                  { id: "THIS_MONTH", label: "هذا الشهر" },
                  { id: "LAST_MONTH", label: "الشهر الماضي" },
                  { id: "THIS_YEAR", label: "هذا العام" },

                  ]}
                  onPress={() => setDateFilter(f.id)}
                  activeOpacity={0.8}
                >

                      styles.filterChipText,
                      dateFilter === f.id && styles.filterChipTextActive,
                    ]}
                  >
                    {f.label}
                  </Text>
                </TouchableOpacity>
              ))}
            </ScrollView>

            {dateFilter === "CUSTOM" && (

import { useRouter } from "expo-router";
import { useState } from "react";
import {
  FlatList,
  StyleSheet,
  Text,
  TouchableOpacity,
  View,
} from "react-native";
import {
  SafeAreaView,
  useSafeAreaInsets,
} from "react-native-safe-area-context";
import InvoiceBottomSheet from "../components/InvoiceBottomSheet";
import { useStore } from "../store/useStore";
import { Invoice } from "../types";

export default function InvoicesScreen() {
  const router = useRouter();
  const { invoices } = useStore();
  const [selectedInvoice, setSelectedInvoice] = useState<Invoice | null>(null);
  const [modalVisible, setModalVisible] = useState(false);

  // Reserve space for the absolutely-positioned tab bar so the FAB and the
  // last invoice card aren't hidden behind it.
  const insets = useSafeAreaInsets();
  const TAB_BAR_HEIGHT = 56;
  const BOTTOM_RESERVED = TAB_BAR_HEIGHT + insets.bottom;

  const openNewInvoice = () => {
    setSelectedInvoice(null);
    setModalVisible(true);
  };

  const openEditInvoice = (invoice: Invoice) => {
    setSelectedInvoice(invoice);
    setModalVisible(true);
  };

  const formatDate = (dateStr: string) => {
    return new Date(dateStr).toLocaleDateString("ar-EG");
  };

  const computeTotal = (invoice: Invoice) => {
    const subtotal = invoice.items.reduce(
      (sum, item) => sum + item.quantity * item.price,
      0,
    );
    const taxAmount = (subtotal * invoice.tax) / 100;
    const discountAmount = (subtotal * invoice.discount) / 100;
    return subtotal + taxAmount - discountAmount;
  };

  return (
    <SafeAreaView style={styles.safeArea} edges={["top"]}>
      <View style={styles.container}>
        <FlatList
          data={invoices}
          keyExtractor={(item) => item.id}
          contentContainerStyle={[
            styles.listContent,
            { paddingBottom: 32 + BOTTOM_RESERVED },
          ]}
          renderItem={({ item }) => (
            <TouchableOpacity
              style={styles.card}
              onPress={() => router.push(`/invoice/${item.id}`)}
              onLongPress={() => openEditInvoice(item)}
              activeOpacity={0.7}
            >
              <Text style={styles.customerName}>{item.customerName}</Text>
              <Text style={styles.companyName}>{item.companyName}</Text>
              <View style={styles.row}>
                <Text style={styles.typeBadge}>
                  {item.type === "ARABIC_INV"
                    ? "بالعربية (Arabic)"
                    : "بالإنجليزية (English)"}
                </Text>
                <Text style={styles.date}>{formatDate(item.date)}</Text>
                <Text style={styles.total}>
                  الإجمالي: {computeTotal(item).toFixed(2)}
                </Text>
              </View>
            </TouchableOpacity>
          )}
          ListEmptyComponent={
            <Text style={styles.empty}>لا توجد فواتير بعد</Text>

        setStatus("denied");
        return;

          Contacts.Fields.Name,
          Contacts.Fields.FirstName,
          Contacts.Fields.LastName,
          Contacts.Fields.PhoneNumbers,
        ],
        sort: Contacts.SortTypes.FirstName,
        pageSize: 1000,
        pageOffset: 0,
      });

        "[Contacts] Retrieved records:",
        result.data.length,
        "Has next page:",
        result.hasNextPage,
      );

      // Explicitly build ContactCard[] to avoid null type errors.
      const formatted = result.data
        .reduce<ContactCard[]>((acc, contact) => {
          const firstName = contact.firstName?.trim() ?? "";
          const lastName = contact.lastName?.trim() ?? "";

          const name =
            contact.name?.trim() || `${firstName} ${lastName}`.trim();

          const phone = contact.phoneNumbers
            ?.find((item) => item.number?.trim())
            ?.number?.trim();

          // Keep contacts with a name or a phone number.
          if (!name && !phone) {
            return acc;
          }

          const displayName = name || phone || "Unknown contact";

          acc.push({
            id:
              contact.id ??
              `${displayName}-${phone ?? `${firstName}-${lastName}`}`,
            name: displayName,
            phone,
          });

          return acc;
        }, [])
        .sort((a, b) => a.name.localeCompare(b.name));

      console.log("[Contacts] Displayable records:", formatted.length);

      setContacts(formatted);
      setStatus("granted");
    } catch (error) {
      console.error("[Contacts] Loading failed:", error);
      setContacts([]);
      setStatus("error");

  const TAB_BAR_HEIGHT = 56;
  const BOTTOM_RESERVED = TAB_BAR_HEIGHT + insets.bottom;

  const project = projects.find((p) => p.id === id);

  const filteredTransactions = useMemo(() => {
    return transactions
      .filter((t) => t.projectId === id)
      .filter((t) => {
        if (!timeFilter || timeFilter === "ALL") return true;

        const txTime = new Date(t.date).getTime();
        const now = new Date();

        if (timeFilter === "THIS_MONTH") {
          const start = new Date(
            now.getFullYear(),
            now.getMonth(),
            1,
          ).getTime();
          return txTime >= start;
        }
        if (timeFilter === "LAST_MONTH") {
          const start = new Date(
            now.getFullYear(),
            now.getMonth() - 1,
            1,
          ).getTime();
          const end = new Date(
            now.getFullYear(),
            now.getMonth(),
            0,
            23,
            59,
            59,
            999,
          ).getTime();
          return txTime >= start && txTime <= end;
        }
        if (timeFilter === "THIS_YEAR") {
          const start = new Date(now.getFullYear(), 0, 1).getTime();
          return txTime >= start;
        }
        return true;
      });
  }, [id, timeFilter, transactions]);

  const periodLabel = useMemo(() => {
    let label = "تاريخ المشروع بالكامل";
    if (timeFilter === "THIS_MONTH") label = "هذا الشهر";
    if (timeFilter === "LAST_MONTH") label = "الشهر الماضي";
    if (timeFilter === "THIS_YEAR") label = "هذا العام";
    return label;
  }, [timeFilter]);

  useEffect(() => {
    if (!project) return;

    let cancelled = false;

    const generateHtml = async () => {
      setIsRegenerating(true);
      try {
        const html = await getArFinancialReportHtml(
          project,
          filteredTransactions,
          periodLabel,
          reportSize,
        );
        if (!cancelled) setHtmlContent(html);
      } finally {
        if (!cancelled) setIsRegenerating(false);
      }

    generateHtml();
    return () => {
      cancelled = true;
    };
  }, [project, filteredTransactions, periodLabel, reportSize]);

const PALETTE = [
  "#10b981", // green (income)
  "#8B5CF6", // violet (loan received)
  "#EF4444", // red (outflow)
  "#2563eb", // blue (transfer)
  "#f59e0b", // amber
  "#0ea5e9", // sky

  "#64748b", // slate

export default function ProjectSettingsScreen() {
  const { id } = useLocalSearchParams<{ id: string }>();

  const {
    projects,
    transactions,
    addTransactionType,
    updateTransactionType,
    removeTransactionType,
    reorderTransactionTypes,
  } = useStore();

  const project = projects.find((p) => p.id === id);

  // -------- Add form state --------
  const [newLabel, setNewLabel] = useState("");
  const [newDirection, setNewDirection] =
    useState<TransactionDirection>("OUTFLOW");
  const [newColor, setNewColor] = useState(PALETTE[2]);

  // -------- Inline edit state --------
  const [editingId, setEditingId] = useState<string | null>(null);
  const [editLabel, setEditLabel] = useState("");
  const [editColor, setEditColor] = useState<string>("");
  const [editDirection, setEditDirection] =
    useState<TransactionDirection>("OUTFLOW");

  // Map of type id -> count of transactions, computed once per render pass.
  const usageCounts = useMemo(() => {
    if (!project) return {} as Record<string, number>;
    const counts: Record<string, number> = {};
    transactions.forEach((t) => {
      if (t.projectId !== project.id) return;
      counts[t.type] = (counts[t.type] ?? 0) + 1;
    });
    return counts;
  }, [transactions, project]);

  if (!project) {
    return (
      <View style={styles.center}>
        <Text>المشروع غير موجود</Text>
      </View>
    );
  }

  // Generate a stable id for a new custom type. Uppercased slug + timestamp
  // so the user can re-create a type with the same label later without a
  // collision.
  const makeId = (label: string) => {
    const slug = label
      .trim()
      .toUpperCase()
      .replace(/[^\p{L}\p{N}]+/gu, "_")
      .replace(/^_+|_+$/g, "");
    return `${slug || "TYPE"}_${Date.now().toString(36)}`;
  };

  const handleAdd = () => {
    const label = newLabel.trim();
    if (!label) return;

    // Duplicate label guard (case-insensitive) — cheap and prevents user
    // confusion when two chips read the same.
    const dup = project.transactionTypes.some(
      (t) => t.label.trim() === label && !t.archived,
    );
    if (dup) {
      Alert.alert("نوع مكرر", `يوجد نوع بنفس الاسم "${label}" بالفعل.`);
      return;
    }

    const def: TransactionTypeDef = {
      id: makeId(label),
      label,
      direction: newDirection,
      affectsBalance: newDirection !== "NEUTRAL",
      color: newColor,
    };

    setNewLabel("");
    setNewDirection("OUTFLOW");
    setNewColor(PALETTE[2]);

    setEditLabel("");
    setEditColor("");
    setEditDirection("OUTFLOW");
  };

    const label = editLabel.trim();
    if (!label) return;

    // Duplicate label guard, excluding the row we're editing.
    const dup = project.transactionTypes.some(
      (t) => t.id !== def.id && t.label.trim() === label && !t.archived,
    );
    if (dup) {
      Alert.alert("نوع مكرر", `يوجد نوع بنفس الاسم "${label}" بالفعل.`);
      return;
    }

    updateTransactionType(project.id, def.id, {
      label,
      color: editColor,
      // System types keep their direction — that direction is baked into
      // the loan/report semantics, so let them only change cosmetic fields.
      ...(def.isSystem
        ? {}
        : {
            direction: editDirection,
            affectsBalance: editDirection !== "NEUTRAL",
          }),
    });
    cancelEdit();
  };

    const inUse = (usageCounts[def.id] ?? 0) > 0;

    // not deleted, so historical rows keep a resolvable label.
    if (def.isSystem || inUse) {
      Alert.alert(
        "أرشفة النوع",
        def.isSystem
          ? `"${def.label}" نوع نظامي — سيتم إخفاؤه من القوائم مع الحفاظ على المعاملات القديمة.`
          : `"${def.label}" مستخدم في ${usageCounts[def.id]} معاملة — سيتم إخفاؤه من القوائم بدلاً من حذفه.`,
        [
          { text: "إلغاء", style: "cancel" },
          {
            text: "أرشفة",
            style: "destructive",
            onPress: () => removeTransactionType(project.id, def.id),
          },
        ],
      );
      return;
    }

    Alert.alert("حذف النوع", `هل أنت متأكد من حذف "${def.label}"؟`, [
      { text: "إلغاء", style: "cancel" },
      {
        text: "حذف",
        style: "destructive",
        onPress: () => removeTransactionType(project.id, def.id),
      },
    ]);
  };

    setContactSearch("");

      phone: manualPhone.trim() || undefined,
    });
    setManualName("");
    setManualPhone("");
    setPersonPickerOpen(false);
  };

  const handleSave = () => {
    const amountNum = parseFormattedNumber(txAmount);
    if (!amountNum || isNaN(amountNum)) {
      setErrorMessage("يرجى إدخال مبلغ صحيح.");
      return;
    }

    if (loanType && !person && !txDesc.trim()) {
      setErrorMessage("يرجى اختيار الشخص أو كتابة ملاحظة.");
      return;
    }

    const fallbackDesc = loanType
      ? ""
      : txType === "TRANSFER"
        ? "تحويل داخلي"
        : "بدون وصف";

    const transactionData: Transaction = {
      id: transactionToEdit ? transactionToEdit.id : Date.now().toString(),
      projectId,
      type: txType,
      method: txMethod,
      amount: amountNum,
      description: txDesc || fallbackDesc,
      date: txDate.toISOString(),
      person: loanType && person ? person : undefined,
    };

    if (transactionToEdit) {
      updateTransaction(transactionData);
    } else {
      addTransaction(transactionData);
    }

    setTxDesc("");
    setTxDate(new Date());
    setTxType(defaultTypeId);
    setTxMethod("CASH");
    setPerson(null);
    setShowDatePicker(false);
    setErrorMessage(null);
    setPersonPickerOpen(false);
    onClose();
  };

      base64: false,
    });
    if (!result.canceled && result.assets.length > 0) {
      setCompanyLogo(result.assets[0].uri);
    }
  };

  const addItem = () => {
    const newItem = {
      id: generateId(),
      item: "",
      description: "",
      quantity: 1,
      price: 0,
    };
    setItems([...items, newItem]);
    setPriceInputs((prev) => ({ ...prev, [newItem.id]: "" }));
  };

  const removeItem = (id: string) => {
    if (items.length === 1) return;
    setItems(items.filter((item) => item.id !== id));
    setPriceInputs((prev) => {
      const newInputs = { ...prev };
      delete newInputs[id];
      return newInputs;
    });
  };

  const updateItem = (id: string, field: keyof InvoiceItem, value: any) => {
    setItems(
      items.map((item) =>
        item.id === id ? { ...item, [field]: value } : item,
      ),
    );
  };

  const handlePriceChange = (id: string, text: string) => {
    setPriceInputs((prev) => ({ ...prev, [id]: text }));
    const num = parseFormattedNumberWithDecimals(text);
    if (!isNaN(num) && text !== "" && !text.endsWith(".")) {
      updateItem(id, "price", num);
    } else if (text === "" || text === "-") {
      updateItem(id, "price", 0);
    }
  };
  const handlePaidAmountChange = (text: string) => {
    setPaidAmountInput(text);
    const num = parseFormattedNumberWithDecimals(text);
    if (!isNaN(num) && text !== "" && !text.endsWith(".")) {
      setPaidAmount(num);
    } else if (text === "" || text === "-") {
      setPaidAmount(0);
    }
  };
  const handlePhoneChange = (text: string) => {
    // Keep only digits
    const digits = text.replace(/\D/g, "");
    setPhoneNumber(digits);
  };

  const handleDateChange = (event: any, selectedDate?: Date) => {
    if (Platform.OS === "android") setShowDatePicker(false);
    if (selectedDate) setDate(selectedDate);
  };

  const handleTimeChange = (event: any, selectedTime?: Date) => {
    if (Platform.OS === "android") setShowTimePicker(false);
    if (selectedTime) setTime(selectedTime);
  };

  const computeSubtotal = () => {
    return items.reduce((sum, item) => sum + item.quantity * item.price, 0);
  };
  const subtotal = computeSubtotal();
  const taxAmount = (subtotal * tax) / 100;
  const discountAmount = (subtotal * discount) / 100;
  const total = subtotal + taxAmount - discountAmount;

  // Display formatted phone
  const displayPhone = formatPhoneNumber(phoneNumber);

  return (
    <Modal visible={visible} animationType="slide" transparent>
      <KeyboardAvoidingView
        behavior={Platform.OS === "ios" ? "padding" : "height"}
        style={styles.modalOverlay}
      >
        <View style={styles.modalContent}>
          {/* Sheet Handle */}
          <View style={styles.sheetHandleContainer}>
            <View style={styles.sheetHandle} />
          </View>

          {/* Top Bar */}
          <View style={styles.headerBar}>
            <TouchableOpacity onPress={onClose} style={styles.closeBtn}>
              <Text style={styles.closeBtnText}>✕</Text>
            </TouchableOpacity>
            <Text style={styles.modalHeader}>
              {invoiceToEdit ? "تعديل الفاتورة" : "إضافة فاتورة جديدة"}
            </Text>
          </View>

          <ScrollView
            showsVerticalScrollIndicator={false}
            contentContainerStyle={styles.scrollContent}
          >
            {/* Validation Error Banner */}
            {errorMessage && (
              <View style={styles.errorBanner}>
                <Text style={styles.errorBannerText}>{errorMessage}</Text>
              </View>
            )}

            {/* Type Switcher */}
            <View style={styles.segmentedControl}>
              <TouchableOpacity
                style={[
                  styles.segmentBtn,
                  type === "ARABIC_INV" && styles.segmentBtnActive,
                ]}
                onPress={() => setType("ARABIC_INV")}
              >
                <Text
                  style={[
                    styles.segmentText,
                    type === "ARABIC_INV" && styles.segmentTextActive,
                  ]}
                >
                  بالعربي (Arabic)
                </Text>
              </TouchableOpacity>
              <TouchableOpacity
                style={[
                  styles.segmentBtn,
                  type === "ENGLISH_INV" && styles.segmentBtnActive,
                ]}
                onPress={() => setType("ENGLISH_INV")}
              >
                <Text
                  style={[
                    styles.segmentText,
                    type === "ENGLISH_INV" && styles.segmentTextActive,
                  ]}
                >
                  بالإنجليزية (English)
                </Text>
              </TouchableOpacity>
            </View>

            {/* Company Logo Picker */}
            <Text style={styles.fieldLabel}>شعار الشركة</Text>
            <TouchableOpacity style={styles.logoPicker} onPress={pickImage}>
              {companyLogo ? (
                <View style={styles.logoPreviewContainer}>
                  <Image
                    source={{ uri: companyLogo }}
                    style={styles.logoImage}
                  />
                  <Text style={styles.changeLogoText}>تغيير الشعار</Text>
                </View>
              ) : (
                <View style={styles.logoPlaceholderContainer}>
                  <Text style={styles.uploadIcon}>📷</Text>
                  <Text style={styles.logoPlaceholder}>
                    اضغط لرفع شعار الشركة
                  </Text>
                </View>
              )}
            </TouchableOpacity>

            {/* Primary Details Input */}
            <View style={styles.inputGroup}>
              <Text style={styles.fieldLabel}>اسم الشركة</Text>
              <TextInput
                style={styles.input}
                placeholder="أدخل اسم الشركة"
                placeholderTextColor="#9CA3AF"
                value={companyName}
                onChangeText={setCompanyName}
                textAlign="right"
              />
            </View>

            {/* Invoice number Input */}
            <View style={styles.inputGroup}>
              <Text style={styles.fieldLabel}>رقم الفاتورة</Text>
              <TextInput
                style={styles.input}
                placeholder="أدخل رقم الفاتورة"
                placeholderTextColor="#9CA3AF"
                value={invoiceNumber}
                onChangeText={setInvoiceNumber}
                textAlign="right"
              />
            </View>

            {/* Company Phone */}
            <View style={styles.inputGroup}>
              <Text style={styles.fieldLabel}>رقم هاتف الشركة (اختياري)</Text>
              <TextInput
                style={styles.input}
                placeholder="أدخل رقم هاتف الشركة"
                placeholderTextColor="#9CA3AF"
                value={formatPhoneNumber(companyPhone)}
                onChangeText={(text) =>
                  setCompanyPhone(stripPhoneFormatting(text))
                }
                textAlign="right"
                keyboardType="phone-pad"
                maxLength={14}
              />
            </View>

            {/* Company Address */}
            <View style={styles.inputGroup}>
              <Text style={styles.fieldLabel}>عنوان الشركة (اختياري)</Text>
              <TextInput
                style={[styles.input, styles.textArea]}
                placeholder="أدخل عنوان الشركة"
                placeholderTextColor="#9CA3AF"
                value={companyAddress}
                onChangeText={setCompanyAddress}
                textAlign="right"
                multiline
                numberOfLines={2}
              />
            </View>
            <View style={styles.inputGroup}>
              <Text style={styles.fieldLabel}>اسم العميل</Text>
              <TextInput
                style={styles.input}
                placeholder="أدخل اسم العميل"
                placeholderTextColor="#9CA3AF"
                value={customerName}
                onChangeText={setCustomerName}
                textAlign="right"
              />
            </View>

            {/* Phone Input – now optional */}
            <View style={styles.inputGroup}>
              <Text style={styles.fieldLabel}>رقم الهاتف (اختياري)</Text>
              <TextInput
                style={styles.input}
                placeholder="(XXX) XXX-XXXX"
                placeholderTextColor="#9CA3AF"
                value={displayPhone}
                onChangeText={handlePhoneChange}
                textAlign="right"
                keyboardType="phone-pad"
                maxLength={14}
              />
            </View>

            <View style={styles.inputGroup}>
              <Text style={styles.fieldLabel}>العنوان *</Text>
              <TextInput
                style={[styles.input, styles.textArea]}
                placeholder="أدخل العنوان"
                placeholderTextColor="#9CA3AF"
                value={address}
                onChangeText={setAddress}
                textAlign="right"
                multiline
                numberOfLines={2}
              />
            </View>
            <View
              style={[styles.inputGroup, { marginTop: 14, marginBottom: 0 }]}
            >
              <Text style={styles.fieldLabel}>المبلغ المدفوع</Text>
              <TextInput
                style={styles.input}
                placeholder="0"
                placeholderTextColor="#9CA3AF"
                keyboardType="decimal-pad"
                value={formatPriceInputDisplay(paidAmountInput)}
                onChangeText={handlePaidAmountChange}
                textAlign="right"
              />
            </View>

            {paidAmount > 0 && (
              <View style={styles.summaryRow}>
                <Text
                  style={[
                    styles.summaryValue,
                    {
                      fontWeight: "800",
                      color: total - paidAmount >= 0 ? "#059669" : "#DC2626",
                    },
                  ]}
                >
                  {formatNumberWithCommas(total - paidAmount)}
                </Text>
                <Text
                  style={[
                    styles.summaryLabel,
                    { fontWeight: "700", color: "#0F172A" },
                  ]}
                >
                  المتبقي
                </Text>
              </View>
            )}

            {/* ---------- DATE + TIME ---------- */}
            <View style={styles.inputGroup}>
              <Text style={styles.fieldLabel}>تاريخ الفاتورة</Text>
              <TouchableOpacity
                style={styles.dateInput}
                onPress={() => setShowDatePicker(true)}
              >
                <Text style={styles.dateIcon}>📅</Text>
                <Text style={styles.dateText}>
                  {date.toISOString().split("T")[0]}
                </Text>
              </TouchableOpacity>
            </View>

            {showDatePicker && (
              <DateTimePicker
                value={date}
                mode="date"
                display="default"
                onChange={handleDateChange}
              />
            )}

            <View style={styles.inputGroup}>
              <Text style={styles.fieldLabel}>وقت الفاتورة</Text>
              <TouchableOpacity
                style={styles.dateInput}
                onPress={() => setShowTimePicker(true)}
              >
                <Text style={styles.dateIcon}>🕒</Text>
                <Text style={styles.dateText}>{formatTimeLabel(time)}</Text>
              </TouchableOpacity>
            </View>

            {showTimePicker && (
              <DateTimePicker
                value={time}
                mode="time"
                is24Hour={false}
                display={Platform.OS === "ios" ? "spinner" : "default"}
                onChange={handleTimeChange}
              />
            )}

            {Platform.OS === "ios" && showTimePicker && (
              <TouchableOpacity
                style={styles.iosDoneBtn}
                onPress={() => setShowTimePicker(false)}
              >
                <Text style={styles.iosDoneBtnText}>تم</Text>
              </TouchableOpacity>
            )}

            {/* Items Section */}
            <View style={styles.sectionHeaderRow}>
              <Text style={styles.sectionTitle}>بنود الفاتورة</Text>
              <Text style={styles.itemCountBadge}>{items.length} بنود</Text>
            </View>

            {items.map((item, index) => (
              <View key={item.id} style={styles.itemCard}>
                <View style={styles.itemCardHeader}>
                  <TouchableOpacity
                    style={styles.removeItemBtn}
                    onPress={() => removeItem(item.id)}
                    disabled={items.length === 1}
                  >
                    <Text
                      style={[
                        styles.removeItemBtnText,
                        items.length === 1 && styles.disabledRemoveBtn,
                      ]}
                    >
                      حذف البند
                    </Text>
                  </TouchableOpacity>
                  <Text style={styles.itemCardTitle}>بند #{index + 1}</Text>
                </View>

                <TextInput
                  style={[styles.input, styles.itemInput]}
                  placeholder="رقم القطعة"
                  placeholderTextColor="#9CA3AF"
                  value={item.item}
                  onChangeText={(val) => updateItem(item.id, "item", val)}
                  textAlign="right"
                />

                <TextInput
                  style={[styles.input, styles.itemInput]}
                  placeholder="الوصف التفصيلي (اختياري)"
                  placeholderTextColor="#9CA3AF"
                  value={item.description}
                  onChangeText={(val) =>
                    updateItem(item.id, "description", val)
                  }
                  textAlign="right"
                />

                <View style={styles.itemNumbersRow}>
                  <View style={styles.flexInputContainer}>
                    <Text style={styles.subFieldLabel}>السعر</Text>
                    <TextInput
                      style={styles.input}
                      placeholder="0.00"
                      placeholderTextColor="#9CA3AF"
                      keyboardType="decimal-pad"
                      value={formatPriceInputDisplay(
                        priceInputs[item.id] ?? "",
                      )}
                      onChangeText={(text) => handlePriceChange(item.id, text)}
                      textAlign="right"
                    />
                  </View>

                  <View style={styles.flexInputContainer}>
                    <Text style={styles.subFieldLabel}>الكمية</Text>
                    <TextInput
                      style={styles.input}
                      placeholder="1"
                      placeholderTextColor="#9CA3AF"
                      keyboardType="numeric"
                      value={String(item.quantity)}
                      onChangeText={(val) =>
                        updateItem(item.id, "quantity", Number(val) || 0)
                      }
                      textAlign="right"
                    />
                  </View>
                </View>

                <View style={styles.lineTotalRow}>
                  <Text style={styles.lineTotalVal}>
                    {formatNumberWithCommas(item.quantity * item.price)}
                  </Text>
                  <Text style={styles.lineTotalLabel}>الإجمالي الجزئي:</Text>
                </View>
              </View>
            ))}

            <TouchableOpacity style={styles.addItemBtn} onPress={addItem}>
              <Text style={styles.addItemBtnText}>+ إضافة بند جديد</Text>
            </TouchableOpacity>

            {/* Calculations & Totals Card */}
            <View style={styles.calculationCard}>
              <View style={styles.totalsInputRow}>
                <View style={styles.totalsInputGroup}>
                  <Text style={styles.subFieldLabel}>الخصم (%)</Text>
                  <TextInput
                    style={styles.input}
                    keyboardType="numeric"
                    value={String(discount)}
                    onChangeText={(val) => setDiscount(Number(val) || 0)}
                    textAlign="center"
                  />
                </View>

                <View style={styles.totalsInputGroup}>
                  <Text style={styles.subFieldLabel}>الضريبة (%)</Text>
                  <TextInput
                    style={styles.input}
                    keyboardType="numeric"
                    value={String(tax)}
                    onChangeText={(val) => setTax(Number(val) || 0)}
                    textAlign="center"
                  />
                </View>
              </View>

              <View style={styles.divider} />

              <View style={styles.summaryRow}>
                <Text style={styles.summaryValue}>
                  {formatNumberWithCommas(subtotal)}
                </Text>
                <Text style={styles.summaryLabel}>المجموع الفرعي</Text>
              </View>
              <View style={styles.summaryRow}>
                <Text style={styles.summaryValue}>
                  {formatNumberWithCommas(taxAmount)}
                </Text>
                <Text style={styles.summaryLabel}>قيمة الضريبة</Text>
              </View>
              <View style={styles.summaryRow}>
                <Text style={styles.summaryValue}>
                  -{formatNumberWithCommas(discountAmount)}
                </Text>
                <Text style={styles.summaryLabel}>قيمة الخصم</Text>
              </View>

              <View style={styles.totalRowContainer}>
                <Text style={styles.totalValue}>
                  {formatNumberWithCommas(total)}
                </Text>
                <Text style={styles.totalLabel}>المبلغ الإجمالي</Text>
              </View>
            </View>

            {/* Action Buttons */}
            <View style={styles.actionContainer}>
              <TouchableOpacity style={styles.saveBtn} onPress={handleSave}>
                <Text style={styles.saveBtnText}>
                  {invoiceToEdit ? "حفظ التعديلات" : "إنشاء الفاتورة"}
                </Text>
              </TouchableOpacity>

              {invoiceToEdit && (
                <TouchableOpacity
                  style={styles.deleteBtn}
                  onPress={handleDelete}
                >
                  <Text style={styles.deleteBtnText}>حذف الفاتورة</Text>
                </TouchableOpacity>
              )}
            </View>
          </ScrollView>
        </View>
      </KeyboardAvoidingView>
    </Modal>
  );
}

import { useColorScheme as useRNColorScheme } from 'react-native';

/**
 * To support static rendering, this value needs to be re-calculated on the client side for web
 */
export function useColorScheme() {
  const [hasHydrated, setHasHydrated] = useState(false);

  useEffect(() => {
    setHasHydrated(true);
  }, []);

  const colorScheme = useRNColorScheme();

  if (hasHydrated) {
    return colorScheme;
  }

  return 'light';
}

async function getImageBase64(uri: string): Promise<string> {
  try {
    const base64 = await FileSystem.readAsStringAsync(uri, {
      encoding: FileSystem.EncodingType.Base64,
    });
    return `data:image/png;base64,${base64}`;
  } catch (error) {
    console.warn("Could not read image, using fallback:", error);
    return "";
  }
}

export const getInvoiceHtml = async (invoice: Invoice): Promise<string> => {
  // ----- Logo: try company logo, fallback to default app icon -----
  async function getAssetBase64(asset: Asset): Promise<string> {
    await asset.downloadAsync();
    let uri = asset.localUri || asset.uri;

    if (uri.startsWith("http://") || uri.startsWith("https://")) {
      const localFile = FileSystem.documentDirectory + "temp_logo.png";
      await FileSystem.downloadAsync(uri, localFile);
      uri = localFile;
    }

    try {
      const base64 = await FileSystem.readAsStringAsync(uri, {
        encoding: FileSystem.EncodingType.Base64,
      });
      return base64;
    } catch (readError) {
      console.warn("Reading asset failed, attempting to copy:", readError);
      const fallbackPath = FileSystem.documentDirectory + "fallback_logo.png";
      await FileSystem.copyAsync({ from: uri, to: fallbackPath });
      const base64 = await FileSystem.readAsStringAsync(fallbackPath, {
        encoding: FileSystem.EncodingType.Base64,
      });
      return base64;
    }
  }
  let logoDataUri = "";
  if (invoice.companyLogo) {
    try {
      logoDataUri = await getImageBase64(invoice.companyLogo);
    } catch (e) {
      console.warn("Failed to load company logo, using default.", e);
    }
  }
  if (!logoDataUri) {
    const defaultAsset = Asset.fromModule(require("@/assets/images/icon.png"));
    const base64 = await getAssetBase64(defaultAsset);
    logoDataUri = `data:image/png;base64,${base64}`;
  }

  // ----- Calculate totals -----
  const subtotal = invoice.items.reduce(
    (sum, item) => sum + item.quantity * item.price,
    0,
  );
  const taxAmount = (subtotal * invoice.tax) / 100;
  const discountAmount = (subtotal * invoice.discount) / 100;
  const total = subtotal + taxAmount - discountAmount;

  // ----- Choose formatter based on invoice type -----
  let formatMoney: (val: number) => string;
  if (invoice.type === "ENGLISH_INV") {
    formatMoney = formatCurrencyPdfWithCents;
  } else {
    formatMoney = formatCurrencyPdf;
  }

  const dateStr = new Date(invoice.date).toLocaleDateString("en-US", {
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  });
  const generatedDate = new Date().toLocaleDateString("en-US", {
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  });

  const escapeHtml = (value: string | undefined | null) => {
    if (!value) return "";
    return String(value)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#039;");
  };

  if (invoice.type === "ARABIC_INV") {
    return getArabicInvoiceHtml(
      invoice,
      logoDataUri,
      formatMoney,
      escapeHtml,
      dateStr,
      generatedDate,
      subtotal,
      taxAmount,
      discountAmount,
      total,
    );
  } else {
    return getEnglishInvoiceHtml(
      invoice,
      logoDataUri,
      formatMoney,
      escapeHtml,
      dateStr,
      generatedDate,
      subtotal,
      taxAmount,
      discountAmount,
      total,
    );
  }
};

export const exportPdf = async (
  html: string,
  customFileName: string = "report.pdf",
  options?: { width: number; height: number },
) => {
  try {
    const { uri } = await Print.printToFileAsync({
      html,
      // expo-print accepts width/height in points for custom page size
      ...(options ? { width: options.width, height: options.height } : {}),
    });
    const safeFileName = customFileName.replace(/[:/\\?%*|"<>]/g, "-");

    const targetDirectory = uri.substring(0, uri.lastIndexOf("/") + 1);
    const newUri = `${targetDirectory}${safeFileName}`;

    await FileSystem.copyAsync({ from: uri, to: newUri });

    await Sharing.shareAsync(newUri, {
      UTI: ".pdf",
      mimeType: "application/pdf",
    });
  } catch (error) {
    console.error("Error exporting PDF:", error);
  }
};

import { LoanPerson, Project, Transaction } from "@/types";

const LOGO_ASSET = require("@/assets/images/icon.png");

export type ReportSize = "A4" | "MOBILE";

async function getAssetBase64(asset: Asset): Promise<string> {
  await asset.downloadAsync();
  let uri = asset.localUri || asset.uri;

  if (uri.startsWith("http://") || uri.startsWith("https://")) {
    const localFile = FileSystem.documentDirectory + "temp_logo.png";
    await FileSystem.downloadAsync(uri, localFile);
    uri = localFile;
  }

  try {
    const base64 = await FileSystem.readAsStringAsync(uri, {
      encoding: FileSystem.EncodingType.Base64,
    });
    return base64;
  } catch (readError) {
    console.warn("Reading asset failed, attempting to copy:", readError);
    const fallbackPath = FileSystem.documentDirectory + "fallback_logo.png";
    await FileSystem.copyAsync({ from: uri, to: fallbackPath });
    const base64 = await FileSystem.readAsStringAsync(fallbackPath, {
      encoding: FileSystem.EncodingType.Base64,
    });
    return base64;
  }
}

const escapeHtml = (value: string | undefined | null) => {
  if (!value) return "";
  return String(value)
    .replace(/&/g, "&amp;")
    .replace(/</g, "&lt;")
    .replace(/>/g, "&gt;")
    .replace(/"/g, "&quot;")
    .replace(/'/g, "&#039;");
};

// ------------------------------------------------------------------
// Loan aggregation
// ------------------------------------------------------------------

type LoanRow = {
  key: string;
  name: string;
  phone?: string;
  given: number;
  received: number;
};

function aggregateLoans(
  project: Project,
  projectTransactions: Transaction[],
): { rows: LoanRow[]; totalGiven: number; totalReceived: number } {
  const byKey = new Map<string, LoanRow>();

  const getPersonKey = (person: LoanPerson | undefined): string | null => {
    if (!person) return null;
    const trimmedName = person.name?.trim();
    if (person.id) return person.id;
    if (!trimmedName) return null;
    return `name:${trimmedName.toLowerCase()}`;
  };

  projectTransactions.forEach((t) => {
    const def = project.transactionTypes.find((d) => d.id === t.type);
    if (!def) return;

    const isGiven = !!def.countsAsLoanGiven;
    const isReceived = !!def.countsAsLoanReceived;
    if (!isGiven && !isReceived) return;

    const personKey = getPersonKey(t.person);
    if (!personKey) return;

    const existing = byKey.get(personKey) ?? {
      key: personKey,
      name: t.person?.name?.trim() || "بدون اسم",
      phone: t.person?.phone,
      given: 0,
      received: 0,
    };

    if (isGiven) existing.given += t.amount;
    if (isReceived) existing.received += t.amount;

    if (t.person?.phone) existing.phone = t.person.phone;
    if (t.person?.name?.trim()) existing.name = t.person.name.trim();

    byKey.set(personKey, existing);
  });

  const rows = Array.from(byKey.values()).sort(
    (a, b) => b.given + b.received - (a.given + a.received),
  );

  const totalGiven = rows.reduce((sum, r) => sum + r.given, 0);
  const totalReceived = rows.reduce((sum, r) => sum + r.received, 0);

  return { rows, totalGiven, totalReceived };
}

// ---------------------------------------------------------------------------
// Helpers: resolve a transaction type definition from a project's catalog.
// These replace the old hardcoded `type === "INCOME"` checks. That means any
// custom type a user creates automatically gets the right balance math.
// ---------------------------------------------------------------------------

export const resolveType = (
  project: Project | undefined,
  typeId: string,
): TransactionTypeDef | undefined =>
  project?.transactionTypes?.find((t) => t.id === typeId);

const isInflowByDef = (def: TransactionTypeDef | undefined): boolean =>
  def?.direction === "INFLOW";

/**
 * Applies a transaction to a project's cash/bank balances.
 * sign = +1 to add, -1 to reverse (used on update/remove).
 *
 * - Unknown types and NEUTRAL / affectsBalance=false types leave balances alone.
 * - TRANSFER flips cash <-> bank depending on `method`.
 * - Everything else moves by direction.
 */
const applyTransaction = (
  project: Project,
  tx: Transaction,
  sign: 1 | -1,
): Project => {
  const def = resolveType(project, tx.type);
  if (!def || !def.affectsBalance) return project;

  const amount = tx.amount * sign;

  if (def.id === "TRANSFER") {
    const isToBank = tx.method === "BANK";
    return {
      ...project,
      currentBank: project.currentBank + (isToBank ? amount : -amount),
      currentCash: project.currentCash + (isToBank ? -amount : amount),
    };
  }

  const delta = isInflowByDef(def) ? amount : -amount;

  return {
    ...project,
    currentCash:
      tx.method === "CASH" ? project.currentCash + delta : project.currentCash,
    currentBank:
      tx.method === "BANK" ? project.currentBank + delta : project.currentBank,
  };
};

              : p,
          ),
        })),

      removeTransactionType: (projectId, typeId) =>
        set((state) => {
          const project = state.projects.find((p) => p.id === projectId);
          const def = project?.transactionTypes.find((t) => t.id === typeId);
          if (!def) return state;

          const inUse = state.transactions.some(
            (t) => t.projectId === projectId && t.type === typeId,
          );

          // System types and types still in use are archived, never removed.

          if (def.isSystem || inUse) {

              projects: state.projects.map((p) =>
                p.id === projectId
                  ? {
                      ...p,
                      transactionTypes: p.transactionTypes.map((t) =>
                        t.id === typeId ? { ...t, archived: true } : t,
                      ),
                    }
                  : p,
              ),
            };

            projects: state.projects.map((p) =>
              p.id === projectId

                    ...p,
                    transactionTypes: p.transactionTypes.filter(

      addMessage: (conversationId, msg) =>
        set((state) => {
          const currentMsgs = state.messages[conversationId] || [];
          const updatedMsgs = [msg, ...currentMsgs];

          const updatedConversations = state.conversations.map((conv) => {
            if (conv.id === conversationId) {
              return {
                ...conv,
                lastMessage: {
                  text: msg.text,
                  timestamp: msg.timestamp,
                  senderId: msg.senderId,
                },
                updatedAt: msg.timestamp,
              };
            }
            return conv;
          });

          return {
            messages: {
              ...state.messages,
              [conversationId]: updatedMsgs,
            },
            conversations: updatedConversations,
          };
        }),
    }),
    {
      name: "finance-storage",
      storage: createJSONStorage(() => AsyncStorage),
      // Bumped to 2 because the Project schema now carries `transactionTypes`.
      version: 2,
      // Backfill defaults for projects persisted before this change.
      migrate: (persistedState: any, _version) => {
        const state = persistedState as Partial<AppState>;
        const projects = (state.projects ?? []).map((p: any) => {
          if (Array.isArray(p.transactionTypes) && p.transactionTypes.length) {
            return p as Project;
          }
          return {
            ...p,
            transactionTypes: DEFAULT_TRANSACTION_TYPES.map((t) => ({
              ...t,
            })),
          } as Project;
        });
        return { ...state, projects };
      },
    },
  ),
);

export const formatRelativeTime = (date: Date | string): string => {
  // ... existing implementation (unchanged)
  const now = new Date();
  const past = typeof date === "string" ? new Date(date) : date;
  const diffMs = now.getTime() - past.getTime();
  const diffSec = Math.floor(diffMs / 1000);
  const diffMin = Math.floor(diffSec / 60);
  const diffHour = Math.floor(diffMin / 60);
  const diffDay = Math.floor(diffHour / 24);

  if (diffDay > 7) {
    return past.toLocaleDateString("en-US", { month: "short", day: "numeric" });
  } else if (diffDay >= 1) {
    return `${diffDay}d ago`;
  } else if (diffHour >= 1) {
    return `${diffHour}h ago`;
  } else if (diffMin >= 1) {
    return `${diffMin}m ago`;
  } else {
    return "Just now";
  }
};

  return new Intl.NumberFormat("en-US", {
    minimumFractionDigits: 0,
    maximumFractionDigits: 2,

export const formatPriceInputDisplay = (raw: string): string => {
  let clean = raw.replace(/,/g, "");
  const parts = clean.split(".");
  if (parts.length > 2) {
    clean = parts[0] + "." + parts.slice(1).join("");
  }
  const [intPart, decPart] = clean.split(".");
  const formattedInt = intPart.replace(/\B(?=(\d{3})+(?!\d))/g, ",");
  if (decPart !== undefined) {
    return formattedInt + "." + decPart;
  } else if (raw.endsWith(".")) {
    return formattedInt + ".";
  } else {
    return formattedInt;
  }
};

/**
 * Formats a raw string of digits into US phone format: (XXX) XXX-XXXX
 * e.g., "2604823730" → "(260) 482-3730"
 * Returns the formatted string; if empty, returns empty.
 */
export const formatPhoneNumber = (digits: string): string => {
  const cleaned = digits.replace(/\D/g, "");
  if (cleaned.length === 0) return "";
  if (cleaned.length <= 3) {
    return `(${cleaned}`;
  }
  if (cleaned.length <= 6) {
    return `(${cleaned.slice(0, 3)}) ${cleaned.slice(3)}`;
  }
  // limit to 10 digits
  const truncated = cleaned.slice(0, 10);
  return `(${truncated.slice(0, 3)}) ${truncated.slice(3, 6)}-${truncated.slice(6)}`;
};
