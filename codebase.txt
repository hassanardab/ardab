# Welcome to your Expo app 👋

This is an [Expo](https://expo.dev) project created with [`create-expo-app`](https://www.npmjs.com/package/create-expo-app).

## Get started

1. Install dependencies

   ```bash
   npm install
   ```

2. Start the app

   ```bash
   npx expo start
   ```

In the output, you'll find options to open the app in a

- [development build](https://docs.expo.dev/develop/development-builds/introduction/)
- [Android emulator](https://docs.expo.dev/workflow/android-studio-emulator/)
- [iOS simulator](https://docs.expo.dev/workflow/ios-simulator/)
- [Expo Go](https://expo.dev/go), a limited sandbox for trying out app development with Expo

You can start developing by editing the files inside the **app** directory. This project uses [file-based routing](https://docs.expo.dev/router/introduction).

=== ./scripts/reset-project.js ===
#!/usr/bin/env node

/**
 * This script is used to reset the project to a blank state.
 * It deletes or moves the /src and /scripts directories to /example based on user input and creates a new /src/app directory with an index.tsx and _layout.tsx file.
 * You can remove the `reset-project` script from package.json and safely delete this file after running it.
 */

const fs = require("fs");
const path = require("path");
const readline = require("readline");

const root = process.cwd();
const oldDirs = ["src", "scripts"];
const exampleDir = "example";
const newAppDir = "src/app";
const exampleDirPath = path.join(root, exampleDir);

const indexContent = `import { Text, View, StyleSheet } from "react-native";

export default function Index() {
  return (
    <View style={styles.container}>
      <Text>Edit src/app/index.tsx to edit this screen.</Text>
    </View>
  );
}

const styles = StyleSheet.create({
  container: {
    flex: 1,
    alignItems: "center",
    justifyContent: "center",
  },
});
`;

const layoutContent = `import { Stack } from "expo-router";

export default function RootLayout() {
  return <Stack />;
}
`;

const rl = readline.createInterface({
  input: process.stdin,
  output: process.stdout,
});

const moveDirectories = async (userInput) => {
  try {
    if (userInput === "y") {
      // Create the app-example directory
      await fs.promises.mkdir(exampleDirPath, { recursive: true });
      console.log(`📁 /${exampleDir} directory created.`);
    }

    // Move old directories to new app-example directory or delete them
    for (const dir of oldDirs) {
      const oldDirPath = path.join(root, dir);
      if (fs.existsSync(oldDirPath)) {
        if (userInput === "y") {
          const newDirPath = path.join(root, exampleDir, dir);
          await fs.promises.rename(oldDirPath, newDirPath);
          console.log(`➡️ /${dir} moved to /${exampleDir}/${dir}.`);
        } else {
          await fs.promises.rm(oldDirPath, { recursive: true, force: true });
          console.log(`❌ /${dir} deleted.`);
        }
      } else {
        console.log(`➡️ /${dir} does not exist, skipping.`);
      }
    }

    // Create new /src/app directory
    const newAppDirPath = path.join(root, newAppDir);
    await fs.promises.mkdir(newAppDirPath, { recursive: true });
    console.log("\n📁 New /src/app directory created.");

    // Create index.tsx
    const indexPath = path.join(newAppDirPath, "index.tsx");
    await fs.promises.writeFile(indexPath, indexContent);
    console.log("📄 src/app/index.tsx created.");

    // Create _layout.tsx
    const layoutPath = path.join(newAppDirPath, "_layout.tsx");
    await fs.promises.writeFile(layoutPath, layoutContent);
    console.log("📄 src/app/_layout.tsx created.");

    console.log("\n✅ Project reset complete. Next steps:");
    console.log(
      `1. Run \`npx expo start\` to start a development server.\n2. Edit src/app/index.tsx to edit the main screen.\n3. Put all your application code in /src, only screens and layout files should be in /src/app.${
        userInput === "y"
          ? `\n4. Delete the /${exampleDir} directory when you're done referencing it.`
          : ""
      }`
    );
  } catch (error) {
    console.error(`❌ Error during script execution: ${error.message}`);
  }
};

rl.question(
  "Do you want to move existing files to /example instead of deleting them? (Y/n): ",
  (answer) => {
    const userInput = answer.trim().toLowerCase() || "y";
    if (userInput === "y" || userInput === "n") {
      moveDirectories(userInput).finally(() => rl.close());
    } else {
      console.log("❌ Invalid input. Please enter 'Y' or 'N'.");
      rl.close();
    }
  }
);

=== ./src/app/index.tsx ===
import { useRouter } from "expo-router";
import { useState } from "react";
import {
  Image,
  Modal,
  ScrollView,
  StyleSheet,
  Text,
  TextInput,
  TouchableOpacity,
  View,
} from "react-native";
import AddTransactionBottomSheet from "../components/AddTransactionBottomSheet";
import { useStore } from "../store/useStore";
import { Project } from "../types";
import { formatCurrency } from "../utils/format";

export default function DashboardScreen() {
  const router = useRouter();
  const { projects, transactions, addProject, updateProject } = useStore();
  const [newProjectName, setNewProjectName] = useState("");
  const [editingProject, setEditingProject] = useState<Project | null>(null);
  const [editName, setEditName] = useState("");
  const [isModalVisible, setIsModalVisible] = useState(false);
  const [selectedProjectId, setSelectedProjectId] = useState<string | null>(
    null,
  );

  const handleCreateProject = () => {
    if (!newProjectName) return;
    addProject({
      id: Date.now().toString(),
      name: newProjectName,
      initialCash: 0,
      initialBank: 0,
      createdAt: new Date().toISOString(),
    });
    setNewProjectName("");
  };

  const handleEditProject = (project: Project) => {
    setEditingProject(project);
    setEditName(project.name);
  };

  const saveEdit = () => {
    if (editingProject && editName.trim()) {
      updateProject(editingProject.id, { name: editName.trim() });
      setEditingProject(null);
      setEditName("");
    }
  };

  const openTransactionModal = (projectId: string) => {
    setSelectedProjectId(projectId);
    setIsModalVisible(true);
  };

  // Format today's date in Arabic
  const formattedDate = new Intl.DateTimeFormat("ar-EG", {
    weekday: "long",
    day: "numeric",
    month: "long",
  }).format(new Date());

  // Compute net loan balance for a project
  const getLoanBalance = (projectId: string): number => {
    const projectTransactions = transactions.filter(
      (t) => t.projectId === projectId,
    );
    const totalReceived = projectTransactions
      .filter((t) => t.type === "LOAN_RECEIVED")
      .reduce((sum, t) => sum + t.amount, 0);
    const totalGiven = projectTransactions
      .filter((t) => t.type === "LOAN_GIVEN")
      .reduce((sum, t) => sum + t.amount, 0);
    return totalReceived - totalGiven;
  };

  return (
    <ScrollView style={styles.container} showsVerticalScrollIndicator={false}>
      {/* Top Header Bar with Date */}
      <View style={styles.headerContainer}>
        <View style={styles.headerTextGroup}>
          <Text style={styles.dateText}>{formattedDate}</Text>
          <Text style={styles.header}>المشاريع النشطة</Text>
        </View>
        <Image
          source={require("@/assets/images/icon.png")}
          style={styles.logo}
        />
      </View>

      {projects.map((project) => {
        const loanBalance = getLoanBalance(project.id);
        return (
          <TouchableOpacity
            key={project.id}
            style={styles.card}
            onPress={() => router.push(`/project/${project.id}`)}
            onLongPress={() => handleEditProject(project)}
            activeOpacity={0.7}
          >
            <Text style={styles.cardTitle}>{project.name}</Text>
            <View style={styles.balanceRow}>
              <View style={styles.balanceBlock}>
                <Text style={styles.label}>الرصيد النقدي</Text>
                <Text style={styles.cashAmount}>
                  {formatCurrency(project.currentCash)}
                </Text>
              </View>
              <View style={styles.balanceBlock}>
                <Text style={styles.label}>الرصيد البنكي</Text>
                <Text style={styles.bankAmount}>
                  {formatCurrency(project.currentBank)}
                </Text>
              </View>
            </View>

            {/* Loan Balance Row */}
            <View style={styles.loanRow}>
              <Text style={styles.label}>صافي القروض</Text>
              <Text
                style={[
                  styles.loanAmount,
                  loanBalance >= 0 ? styles.positiveLoan : styles.negativeLoan,
                ]}
              >
                {formatCurrency(loanBalance)}
              </Text>
            </View>

            <TouchableOpacity
              style={styles.actionBtn}
              onPress={(e) => {
                e.stopPropagation();
                openTransactionModal(project.id);
              }}
            >
              <Text style={styles.actionBtnText}>+ إضافة معاملة</Text>
            </TouchableOpacity>
          </TouchableOpacity>
        );
      })}

      <View style={styles.form}>
        <TextInput
          style={styles.input}
          placeholder="اسم المشروع الجديد"
          value={newProjectName}
          onChangeText={setNewProjectName}
          textAlign="right"
        />
        <TouchableOpacity
          style={styles.primaryBtn}
          onPress={handleCreateProject}
        >
          <Text style={styles.primaryBtnText}>إنشاء مشروع</Text>
        </TouchableOpacity>
      </View>

      {/* Transaction Modal */}
      {selectedProjectId && (
        <AddTransactionBottomSheet
          projectId={selectedProjectId}
          visible={isModalVisible}
          onClose={() => setIsModalVisible(false)}
        />
      )}

      {/* Edit Project Modal */}
      <Modal
        visible={!!editingProject}
        transparent
        animationType="fade"
        onRequestClose={() => setEditingProject(null)}
      >
        <View style={styles.modalOverlay}>
          <View style={styles.editModalContent}>
            <Text style={styles.modalHeader}>العديل على المشروع </Text>
            <TextInput
              style={styles.input}
              placeholder="اسم المشروع"
              value={editName}
              onChangeText={setEditName}
              textAlign="right"
              autoFocus
            />
            <View style={styles.editModalButtons}>
              <TouchableOpacity
                style={[styles.modalBtn, styles.saveBtn]}
                onPress={saveEdit}
              >
                <Text style={styles.saveBtnText}>حفظ</Text>
              </TouchableOpacity>
              <TouchableOpacity
                style={[styles.modalBtn, styles.cancelBtn]}
                onPress={() => {
                  setEditingProject(null);
                  setEditName("");
                }}
              >
                <Text style={styles.cancelBtnText}>إلغاء</Text>
              </TouchableOpacity>
            </View>
          </View>
        </View>
      </Modal>
    </ScrollView>
  );
}

      (sum, item) => sum + item.quantity * item.price,

          if (!timeFilter || timeFilter === "ALL") return true;

          const txTime = new Date(t.date).getTime();
          const now = new Date();

          if (timeFilter === "THIS_MONTH") {
            const start = new Date(
              now.getFullYear(),
              now.getMonth(),
              1,
            ).getTime();
            return txTime >= start;
          }
          if (timeFilter === "LAST_MONTH") {
            const start = new Date(
              now.getFullYear(),
              now.getMonth() - 1,
              1,
            ).getTime();
            const end = new Date(
              now.getFullYear(),
              now.getMonth(),
              0,
              23,
              59,
              59,
              999,
            ).getTime();
            return txTime >= start && txTime <= end;
          }
          if (timeFilter === "THIS_YEAR") {
            const start = new Date(now.getFullYear(), 0, 1).getTime();
            return txTime >= start;
          }
          return true;
        });

      let periodLabel = "تاريخ المشروع بالكامل";
      if (timeFilter === "THIS_MONTH") periodLabel = "هذا الشهر";
      if (timeFilter === "LAST_MONTH") periodLabel = "الشهر الماضي";
      if (timeFilter === "THIS_YEAR") periodLabel = "هذا العام";

      // 👇 Await the HTML generation
      const html = await getArFinancialReportHtml(
        project,
        filteredTransactions,
        periodLabel,
      );
      setHtmlContent(html);
    };

    generateHtml();
  }, [id, timeFilter, project, transactions]);

import { openBrowserAsync, WebBrowserPresentationStyle } from 'expo-web-browser';
import { type ComponentProps } from 'react';

type Props = Omit<ComponentProps<typeof Link>, 'href'> & { href: Href & string };

export function ExternalLink({ href, ...rest }: Props) {
  return (
    <Link
      target="_blank"
      {...rest}
      href={href}
      onPress={async (event) => {
        if (process.env.EXPO_OS !== 'web') {
          // Prevent the default behavior of linking to the default browser on native.
          event.preventDefault();
          // Open the link in an in-app browser.
          await openBrowserAsync(href, {
            presentationStyle: WebBrowserPresentationStyle.AUTOMATIC,
          });
        }
      }}
    />
  );
}

import DateTimePicker from "@react-native-community/datetimepicker";
import * as ImagePicker from "expo-image-picker";
import { useEffect, useState } from "react";
import {
  Image,
  KeyboardAvoidingView,
  Modal,
  Platform,
  ScrollView,
  StyleSheet,
  Text,
  TextInput,
  TouchableOpacity,
  View,
} from "react-native";
import { useStore } from "../store/useStore";
import { Invoice, InvoiceItem, InvoiceType } from "../types";
import {
  formatNumberWithCommas,
  formatPhoneNumber,
  formatPriceInputDisplay,
  parseFormattedNumberWithDecimals,
  stripPhoneFormatting,
} from "../utils/format";

interface Props {
  visible: boolean;
  onClose: () => void;
  invoiceToEdit?: Invoice | null;
}

const generateId = () =>
  Date.now().toString() + Math.random().toString(36).substring(2, 6);

export default function InvoiceBottomSheet({
  visible,
  onClose,
  invoiceToEdit,
}: Props) {
  const { addInvoice, updateInvoice, deleteInvoice } = useStore();

  const [type, setType] = useState<InvoiceType>("ARABIC_INV");
  const [companyName, setCompanyName] = useState("");
  const [companyLogo, setCompanyLogo] = useState<string | undefined>(undefined);
  const [customerName, setCustomerName] = useState("");
  const [companyPhone, setCompanyPhone] = useState("");
  const [phoneNumber, setPhoneNumber] = useState(""); // store raw digits
  const [address, setAddress] = useState("");
  const [date, setDate] = useState(new Date());
  const [companyAddress, setCompanyAddress] = useState("");
  const [showDatePicker, setShowDatePicker] = useState(false);
  const [errorMessage, setErrorMessage] = useState<string | null>(null);
  const [items, setItems] = useState<InvoiceItem[]>([
    { id: generateId(), item: "", description: "", quantity: 1, price: 0 },
  ]);
  const [tax, setTax] = useState(0);
  const [discount, setDiscount] = useState(0);

  // For price inputs per item
  const [priceInputs, setPriceInputs] = useState<Record<string, string>>({});

  // Load data when editing or opening
  useEffect(() => {
    if (visible) {
      setErrorMessage(null);
      if (invoiceToEdit) {
        setType(invoiceToEdit.type || "ARABIC_INV");
        setCompanyName(invoiceToEdit.companyName);
        setCompanyLogo(invoiceToEdit.companyLogo);
        setCompanyAddress(invoiceToEdit.companyAddress || "");
        setCustomerName(invoiceToEdit.customerName);
        setCompanyPhone(invoiceToEdit.companyPhone || "");
        setPhoneNumber(stripPhoneFormatting(invoiceToEdit.phoneNumber || ""));
        setAddress(invoiceToEdit.address || "");
        setDate(new Date(invoiceToEdit.date));
        const loadedItems = invoiceToEdit.items.map((item) => ({
          ...item,
          id: item.id || generateId(),
        }));
        setItems(loadedItems);
        const inputs: Record<string, string> = {};
        loadedItems.forEach((item) => {
          inputs[item.id] = String(item.price);
        });
        setPriceInputs(inputs);
        setTax(invoiceToEdit.tax);
        setDiscount(invoiceToEdit.discount);
      } else {
        // Reset for new invoice
        setType("ARABIC_INV");
        setCompanyName("");
        setCompanyLogo(undefined);
        setCustomerName("");
        setPhoneNumber("");
        setCompanyAddress("");
        setAddress("");
        setCompanyPhone("");
        setDate(new Date());
        const newItem = {
          id: generateId(),
          item: "",
          description: "",
          quantity: 1,
          price: 0,
        };
        setItems([newItem]);
        setPriceInputs({ [newItem.id]: "" });
        setTax(0);
        setDiscount(0);
      }
    }
  }, [visible, invoiceToEdit]);

  const handleSave = () => {
    // Phone number is now optional – only validate required fields
    if (
      !companyName.trim() ||
      !customerName.trim() ||
      !address.trim() ||
      items.some((i) => !i.item.trim() || i.quantity <= 0 || i.price < 0)
    ) {
      setErrorMessage("يرجى ملء جميع الحقول المطلوبة بشكل صحيح");
      return;
    }

    setErrorMessage(null);
    const rawPhone = phoneNumber.replace(/\D/g, ""); // still stored, but can be empty
    const invoiceData: Invoice = {
      id: invoiceToEdit?.id || generateId(),
      type,
      companyName: companyName.trim(),
      companyLogo,
      customerName: customerName.trim(),
      phoneNumber: rawPhone, // store only digits, may be empty
      address: address.trim(),
      date: date.toISOString(),
      companyPhone: companyPhone.trim() || undefined,
      companyAddress: companyAddress.trim() || undefined,
      items: items.map((item) => ({
        ...item,
        id: item.id || generateId(),
      })),
      tax: tax || 0,
      discount: discount || 0,
    };

    if (invoiceToEdit) {
      updateInvoice(invoiceData.id, invoiceData);
    } else {
      addInvoice(invoiceData);
    }
    onClose();
  };

      deleteInvoice(invoiceToEdit.id);

    }

  const pickImage = async () => {

    }

    if (Platform.OS === "android") setShowDatePicker(false);

  };

    return items.reduce((sum, item) => sum + item.quantity * item.price, 0);

  const taxAmount = (subtotal * tax) / 100;
  const discountAmount = (subtotal * discount) / 100;

      <KeyboardAvoidingView
        behavior={Platform.OS === "ios" ? "padding" : "height"}

          {/* Sheet Handle */}

            <View style={styles.sheetHandle} />

          {/* Top Bar */}
          <View style={styles.headerBar}>
            <TouchableOpacity onPress={onClose} style={styles.closeBtn}>
              <Text style={styles.closeBtnText}>✕</Text>
            </TouchableOpacity>
            <Text style={styles.modalHeader}>
              {invoiceToEdit ? "تعديل الفاتورة" : "إضافة فاتورة جديدة"}
            </Text>
          </View>

          <ScrollView

            {/* Validation Error Banner */}

              <View style={styles.errorBanner}>

            )}

            <View style={styles.segmentedControl}>

                  styles.segmentBtn,

              >

import { useEffect, useState } from 'react';
import { useColorScheme as useRNColorScheme } from 'react-native';

/**
 * To support static rendering, this value needs to be re-calculated on the client side for web
 */
export function useColorScheme() {
  const [hasHydrated, setHasHydrated] = useState(false);

  useEffect(() => {
    setHasHydrated(true);
  }, []);

  const colorScheme = useRNColorScheme();

  if (hasHydrated) {
    return colorScheme;
  }

  return 'light';
}

async function getImageBase64(uri: string): Promise<string> {
  try {
    const base64 = await FileSystem.readAsStringAsync(uri, {

import { formatCurrencyPdfWithCents } from "@/utils/format";

export const getInvoiceHtml = async (invoice: Invoice): Promise<string> => {
  // ----- Logo: try company logo, fallback to default app icon -----
  async function getAssetBase64(asset: Asset): Promise<string> {
    await asset.downloadAsync();
    let uri = asset.localUri || asset.uri;

    if (uri.startsWith("http://") || uri.startsWith("https://")) {
      const localFile = FileSystem.documentDirectory + "temp_logo.png";
      await FileSystem.downloadAsync(uri, localFile);
      uri = localFile;
    }

    try {
      const base64 = await FileSystem.readAsStringAsync(uri, {
        encoding: FileSystem.EncodingType.Base64,
      });
      return base64;
    } catch (readError) {
      console.warn("Reading asset failed, attempting to copy:", readError);
      const fallbackPath = FileSystem.documentDirectory + "fallback_logo.png";
      await FileSystem.copyAsync({ from: uri, to: fallbackPath });
      const base64 = await FileSystem.readAsStringAsync(fallbackPath, {
        encoding: FileSystem.EncodingType.Base64,
      });
      return base64;
    }
  }
  let logoDataUri = "";
  if (invoice.companyLogo) {
    try {
      logoDataUri = await getImageBase64(invoice.companyLogo);
    } catch (e) {
      console.warn("Failed to load company logo, using default.", e);
    }
  }
  if (!logoDataUri) {
    const defaultAsset = Asset.fromModule(require("@/assets/images/icon.png"));
    const base64 = await getAssetBase64(defaultAsset);
    logoDataUri = `data:image/png;base64,${base64}`;
  }

  // ----- Calculate totals -----
  const subtotal = invoice.items.reduce(
    (sum, item) => sum + item.quantity * item.price,
    0,
  );
  const taxAmount = (subtotal * invoice.tax) / 100;
  const discountAmount = (subtotal * invoice.discount) / 100;
  const total = subtotal + taxAmount - discountAmount;

  // ----- Choose formatter based on invoice type -----
  let formatMoney: (val: number) => string;
  if (invoice.type === "ENGLISH_INV") {
    formatMoney = formatCurrencyPdfWithCents; // two decimals
  } else {
    formatMoney = formatCurrencyPdf; // whole dollars (original)
  }

  const dateStr = new Date(invoice.date).toLocaleDateString("en-US", {
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  });
  const generatedDate = new Date().toLocaleDateString("en-US", {
    year: "numeric",
    month: "2-digit",
    day: "2-digit",
  });

  const escapeHtml = (value: string | undefined | null) => {
    if (!value) return "";
    return String(value)
      .replace(/&/g, "&amp;")
      .replace(/</g, "&lt;")
      .replace(/>/g, "&gt;")
      .replace(/"/g, "&quot;")
      .replace(/'/g, "&#039;");
  };

  // Build items rows (only used for ARABIC_INV; ENGLISH_INV builds its own)
  const itemsRows = invoice.items
    .map(
      (item, index) => `
        <tr>
          <td class="line-number">${index + 1}</td>
          <td class="item-description">
            <div class="item-name">${escapeHtml(item.item)}</div>
              ${
                item.description
                  ? `<div class="item-details">${escapeHtml(item.description)}</div>`
                  : ""
              }
            </td>
            <td class="operation">${escapeHtml(item.description) || "-"}</td>
            <td class="units">${item.quantity}</td>
            <td class="unit-price">${formatMoney(item.price)}</td>
            <td class="total-price">${formatMoney(item.quantity * item.price)}</td>
          </tr>
        `,
    )
    .join("");

  // Choose template based on invoice type
  if (invoice.type === "ARABIC_INV") {
    return getArabicInvoiceHtml(
      invoice,
      logoDataUri,
      formatMoney,
      escapeHtml,
      dateStr,
      generatedDate,
      subtotal,
      taxAmount,
      discountAmount,
      total,
    );
  } else {
    // ENGLISH_INV does NOT accept itemsRows – it builds its own internally
    return getEnglishInvoiceHtml(
      invoice,
      logoDataUri,
      formatMoney,
      escapeHtml,
      dateStr,
      generatedDate,
      subtotal,
      taxAmount,
      discountAmount,
      total,
    );
  }
};

export const exportPdf = async (
  html: string,
  customFileName: string = "report.pdf",
) => {
  try {
    const { uri } = await Print.printToFileAsync({ html });
    const safeFileName = customFileName.replace(/[:/\\?%*|"<>]/g, "-");

    const targetDirectory = uri.substring(0, uri.lastIndexOf("/") + 1);
    const newUri = `${targetDirectory}${safeFileName}`;

    await FileSystem.copyAsync({ from: uri, to: newUri });

    await Sharing.shareAsync(newUri, {
      UTI: ".pdf",
      mimeType: "application/pdf",
    });
  } catch (error) {
    console.error("Error exporting PDF:", error);
  }
};

async function getAssetBase64(asset: Asset): Promise<string> {
  await asset.downloadAsync();
  let uri = asset.localUri || asset.uri;

  if (uri.startsWith("http://") || uri.startsWith("https://")) {
    const localFile = FileSystem.documentDirectory + "temp_logo.png";
    await FileSystem.downloadAsync(uri, localFile);
    uri = localFile;
  }

  try {
    const base64 = await FileSystem.readAsStringAsync(uri, {
      encoding: FileSystem.EncodingType.Base64,
    });
    return base64;
  } catch (readError) {
    console.warn("Reading asset failed, attempting to copy:", readError);
    const fallbackPath = FileSystem.documentDirectory + "fallback_logo.png";
    await FileSystem.copyAsync({ from: uri, to: fallbackPath });
    const base64 = await FileSystem.readAsStringAsync(fallbackPath, {
      encoding: FileSystem.EncodingType.Base64,
    });
    return base64;
  }
}

// ----- HELPER: determines if a transaction is an inflow (increases balance) -----
const isInflow = (type: string): boolean => {
  return type === "INCOME" || type === "LOAN_RECEIVED";
};

// ----- HELPER: applies a transaction effect to a project's balances -----
const applyTransaction = (project: Project, tx: Transaction, sign: 1 | -1) => {
  const multiplier = sign; // +1 for add, -1 for remove/reverse
  const amount = tx.amount * multiplier;

  if (tx.type === "TRANSFER") {
    const isToBank = tx.method === "BANK";
    return {
      ...project,
      currentBank: project.currentBank + (isToBank ? amount : -amount),
      currentCash: project.currentCash + (isToBank ? -amount : amount),
    };
  }

  const inflow = isInflow(tx.type);
  const delta = inflow ? amount : -amount;

  return {
    ...project,
    currentCash:
      tx.method === "CASH" ? project.currentCash + delta : project.currentCash,
    currentBank:
      tx.method === "BANK" ? project.currentBank + delta : project.currentBank,
  };
};

        set((state) => ({
          invoices: [...state.invoices, invoice],
        })),
      updateInvoice: (id, updates) =>
        set((state) => ({
          invoices: state.invoices.map((inv) =>
            inv.id === id ? { ...inv, ...updates } : inv,
          ),
        })),
      deleteInvoice: (id) =>
        set((state) => ({
          invoices: state.invoices.filter((inv) => inv.id !== id),
        })),

      addProject: (projectData) =>
        set((state) => ({
          projects: [
            ...state.projects,
            {
              ...projectData,
              currentCash: projectData.initialCash,
              currentBank: projectData.initialBank,
            },
          ],
        })),

      addEmployee: (employee) =>
        set((state) => ({
          employees: [...state.employees, employee],
        })),

      updateProject: (id, updates) =>
        set((state) => ({
          projects: state.projects.map((p) =>
            p.id === id ? { ...p, ...updates } : p,
          ),
        })),

      addTransaction: (transaction) =>
        set((state) => {
          const updatedProjects = state.projects.map((proj) =>
            proj.id === transaction.projectId
              ? applyTransaction(proj, transaction, 1)
              : proj,
          );
          return {
            transactions: [...state.transactions, transaction],
            projects: updatedProjects,
          };
        }),

      updateTransaction: (updatedTransaction) =>
        set((state) => {
          const oldTransaction = state.transactions.find(
            (t) => t.id === updatedTransaction.id,
          );
          if (!oldTransaction) return state;

          // Reverse old transaction
          const afterReverse = state.projects.map((proj) =>
            proj.id === oldTransaction.projectId
              ? applyTransaction(proj, oldTransaction, -1)
              : proj,
          );

          // Apply new transaction
          const finalProjects = afterReverse.map((proj) =>
            proj.id === updatedTransaction.projectId
              ? applyTransaction(proj, updatedTransaction, 1)
              : proj,
          );

          return {
            transactions: state.transactions.map((t) =>
              t.id === updatedTransaction.id ? updatedTransaction : t,
            ),
            projects: finalProjects,
          };
        }),

      removeTransaction: (transactionId) =>
        set((state) => {
          const transaction = state.transactions.find(
            (t) => t.id === transactionId,
          );
          if (!transaction) return state;

          const updatedProjects = state.projects.map((proj) =>
            proj.id === transaction.projectId
              ? applyTransaction(proj, transaction, -1)
              : proj,
          );

          return {
            transactions: state.transactions.filter(
              (t) => t.id !== transactionId,
            ),
            projects: updatedProjects,
          };
        }),

      // ----- Messaging -----
      conversations: defaultConversations,
      messages: defaultMessages,

      setConversations: (convs) => set({ conversations: convs }),

        set((state) => ({ conversations: [conv, ...state.conversations] })),

        set((state) => {

          const updatedMsgs = [msg, ...currentMsgs];

          const updatedConversations = state.conversations.map((conv) => {
            if (conv.id === conversationId) {
              return {
                ...conv,
                lastMessage: {
                  text: msg.text,
                  timestamp: msg.timestamp,
                  senderId: msg.senderId,
                },
                updatedAt: msg.timestamp,
              };
            }
            return conv;
          });

          return {
            messages: {
              ...state.messages,
              [conversationId]: updatedMsgs,
            },
            conversations: updatedConversations,
          };
        }),
    }),
    {
      name: "finance-storage",
      storage: createJSONStorage(() => AsyncStorage),
    },

export const formatRelativeTime = (date: Date | string): string => {
  // ... existing implementation (unchanged)
  const now = new Date();
  const past = typeof date === "string" ? new Date(date) : date;
  const diffMs = now.getTime() - past.getTime();
  const diffSec = Math.floor(diffMs / 1000);
  const diffMin = Math.floor(diffSec / 60);
  const diffHour = Math.floor(diffMin / 60);
  const diffDay = Math.floor(diffHour / 24);

  if (diffDay > 7) {
    return past.toLocaleDateString("en-US", { month: "short", day: "numeric" });
  } else if (diffDay >= 1) {
    return `${diffDay}d ago`;
  } else if (diffHour >= 1) {
    return `${diffHour}h ago`;
  } else if (diffMin >= 1) {
    return `${diffMin}m ago`;
  } else {
    return "Just now";
  }
};

  return isNaN(num) ? 0 : num;
};

export const formatPriceInputDisplay = (raw: string): string => {
  let clean = raw.replace(/,/g, "");
  const parts = clean.split(".");
  if (parts.length > 2) {
    clean = parts[0] + "." + parts.slice(1).join("");
  }
  const [intPart, decPart] = clean.split(".");
  const formattedInt = intPart.replace(/\B(?=(\d{3})+(?!\d))/g, ",");
  if (decPart !== undefined) {
    return formattedInt + "." + decPart;
  } else if (raw.endsWith(".")) {
    return formattedInt + ".";
  } else {
    return formattedInt;
  }
};

// ----- NEW: Phone number formatting -----
/**
 * Formats a raw string of digits into US phone format: (XXX) XXX-XXXX
 * e.g., "2604823730" → "(260) 482-3730"
 * Returns the formatted string; if empty, returns empty.
 */
export const formatPhoneNumber = (digits: string): string => {
  const cleaned = digits.replace(/\D/g, "");
  if (cleaned.length === 0) return "";
  if (cleaned.length <= 3) {
    return `(${cleaned}`;
  }
  if (cleaned.length <= 6) {
    return `(${cleaned.slice(0, 3)}) ${cleaned.slice(3)}`;
  }
  // limit to 10 digits
  const truncated = cleaned.slice(0, 10);
  return `(${truncated.slice(0, 3)}) ${truncated.slice(3, 6)}-${truncated.slice(6)}`;
};

/**
 * Removes all non-digit characters from a phone string.
 */
export const stripPhoneFormatting = (phone: string): string => {
  return phone.replace(/\D/g, "");
};

export const formatCurrencyPdfWithCents = (amount: number): string => {
  return new Intl.NumberFormat("en-US", {
    style: "currency",
    currency: "USD",
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  }).format(amount);
};

export default function ConversationScreen() {
  const { id } = useLocalSearchParams<{ id: string }>();

  const messages = useStore((state) => state.messages[id] ?? EMPTY_MESSAGES);
  const addMessage = useStore((state) => state.addMessage);

  const [inputText, setInputText] = useState("");
  const flatListRef = useRef<FlatList>(null);

  useEffect(() => {
    flatListRef.current?.scrollToOffset({ offset: 0, animated: true });
  }, [messages]);

  const handleSend = () => {
    if (!inputText.trim()) return;
    const newMsg: Message = {
      id: Date.now().toString(),
      conversationId: id,
      senderId: CURRENT_USER_ID,
      text: inputText.trim(),
      timestamp: new Date(),
      status: "sent",
    };
    addMessage(id, newMsg);
    setInputText("");
  };

  const renderItem = ({ item }: { item: Message }) => {
    const isOwn = item.senderId === CURRENT_USER_ID;
    const timeString = new Date(item.timestamp).toLocaleTimeString([], {
      hour: "2-digit",
      minute: "2-digit",
    });

    return (
      <View
        style={[styles.messageRow, isOwn ? styles.ownRow : styles.otherRow]}
      >
        <View
          style={[styles.bubble, isOwn ? styles.ownBubble : styles.otherBubble]}
        >
          <Text
            style={[
              styles.messageText,
              isOwn ? styles.ownText : styles.otherText,
            ]}
          >
            {item.text}
          </Text>
          <View style={styles.footerRow}>
            <Text style={[styles.timestampSmall, isOwn && styles.ownTimestamp]}>
              {timeString}
            </Text>
            {isOwn && (
              <Text style={styles.statusText}>
                {item.status === "read" ? " ✓✓" : " ✓"}
              </Text>
            )}
          </View>
        </View>
      </View>
    );
  };

  return (
    <SafeAreaView style={styles.container}>
      <KeyboardAvoidingView
        behavior={Platform.OS === "ios" ? "padding" : undefined}
        style={styles.flex}
        keyboardVerticalOffset={Platform.OS === "ios" ? 90 : 0}
      >
        <FlatList
          ref={flatListRef}
          data={messages}
          keyExtractor={(item) => item.id}
          renderItem={renderItem}
          inverted
          contentContainerStyle={{ paddingHorizontal: 12, paddingVertical: 16 }}
          showsVerticalScrollIndicator={false}
        />
        <View style={styles.inputContainer}>
          <TextInput
            style={styles.input}
            value={inputText}
            onChangeText={setInputText}
            placeholder="Message..."
            placeholderTextColor="#8e8e93"
            multiline

          >
            <Text style={styles.sendButtonText}>↑</Text>
          </TouchableOpacity>
        </View>
      </KeyboardAvoidingView>
    </SafeAreaView>
  );
}
